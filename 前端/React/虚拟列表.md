# 虚拟列表

> 虚拟列表分为上缓存区、可视区、下缓存区。


`App.jsx:`
``` jsx
import VirtualList from './components/VirtualList.jsx';


const Row = ({ index, style, forwardRef }) => {
  const listItemOdd = {
    backgroundColor: 'lightgray'
  }
  const listItemEven = {
    backgroundColor: 'blue'
  }
  return (
    <div className={index % 2 ? 'list-item-odd' : 'list-item-even'} style={style} ref={forwardRef}>
      {`Row ${index}`}
    </div>
  )
}

export default function App() {
  return (
    <>
      <VirtualList
        className="list"
        height={200}
        width={200}
        itemSize={50}
        itemCount={1000}
      >
        {Row}
      </VirtualList>
    </>
  )
}
```


`VirtualList.jsx:`
``` jsx 
import {
    Children,
    useState,
} from 'react';

const VirtualList = (props) => {
    const { height, width, itemSize, itemCount, children: Child } = props;

    const [scroolOffset, setScroolOffset] = useState(0);

    const containerStyle = {
        position: 'relative',
        width,
        height,
        overflow: 'auto',
    }

    const contentStyle = {
        height: itemSize * itemCount,
        width: '100%',
    }

    const CurrentChildren = () => {
        const startIndex = Math.floor(scroolOffset / itemSize);  // 可视范围元素的第一个索引
        const topIndex = Math.max(startIndex - 2, 0);   // 上缓存区
        const numVisibleCount = Math.ceil(height / itemSize);  // 可视范围元素的个数
        const endIndex = Math.min(itemCount, startIndex + numVisibleCount + 2);  // 下缓存区
        const items = [];

        for (let i = topIndex; i < endIndex; i++) {
            const itemStyle = {
                position: 'absolute',
                height: itemSize,
                width: '100%',
                top: itemSize * i,
            }
            items.push(
                <Child key={i} index={i} style={itemStyle} />
            );
        }
        return items;
    }

    const handleScroll = (e) => {
        const { scrollTop } = e.currentTarget;
        setScroolOffset(scrollTop);
    }


    return (
        <>
            <div style={containerStyle} onScroll={handleScroll}>
                <div style={contentStyle}>
                    {CurrentChildren()}
                </div>
            </div>
        </>
    )
};

export default VirtualList;
```
