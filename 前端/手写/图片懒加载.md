# 图片懒加载


`1.html:`
``` html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Document</title>
    <style>
        img {
            width: 100%;
            height: auto;
        }
        .main {
            display: flex;
            flex-direction: column;
            align-items: center;
            justify-content: center;
        }
        .gap {
            width: 500px;
            height: 500px;
            background-color: #f0f0f0;
        }
        .container {
            width: 700px;
            height: 500px;
            overflow: auto;
        }
    </style>
</head>
<body>
    <div class="main">
        <div class="container">
            <div class="gap">1</div>
            <div class="gap">2</div>
            <div class="gap">3</div>
            <div class="gap">4</div>
            <div class="gap">5</div>
            <img class="lazyImg" src="1.jpg" data-src="https://lf3-ea.bytetos.com/obj/eaoffice/ies/bytedance_official/_next/static/images/0-390b5def140dc370854c98b8e82ad394.png">
            <div class="gap">1</div>
            <div class="gap">2</div>
            <div class="gap">3</div>
            <img class="lazyImg" src="1.jpg" data-src="https://lf3-ea.bytetos.com/obj/eaoffice/ies/bytedance_official/_next/static/images/cod-cb0d883e8d2c1c45bafb93eb7cd64468.png">
            <div class="gap">4</div>
            <div class="gap">5</div>
        </div>
    </div>

    <script>
        const observer = new IntersectionObserver(entries => {
            // entries 是一个数组，包含了所有被触发状态变化的节点
            entries.forEach(entry => { 
                if (entry.isIntersecting) {
                    const img = entry.target;
                    img.src = img.dataset.src;
                    console.log('图片懒加载成功！');
                    // 任务完成，停止观察这个特定的 img
                    observer.unobserve(img);
                }
            });
        }, {
            // 监听 .container 内部的滚动
            root: document.querySelector('.container'),
            threshold: 0
        });
        document.querySelectorAll('.lazyImg').forEach(img => observer.observe(img));
    </script>
</body>
</html>
```
