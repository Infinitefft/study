# 拷贝

## 浅拷贝

> 只拷贝第一层，如果遇到还有对象，那么则会变成引用

``` js
const shallowClone = (obj) => {
  if (typeof obj !== 'object') return;

  let res = {};

  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      res[key] = obj[key];
    }
  }
  return res;
}
```



### 深拷贝

> 可以完全拷贝成一份独立的，如果遇到对象的话会对这个对象进行递归做相同的操作


`初级版：不支持正则、时间的拷贝`
``` js
const deepClone = (obj) => {
  if (typeof obj !== 'object') return;

  let newObj = obj instanceof Array ? [] : {};
  for (let key in obj) {
    if (obj.hasOwnProperty(key)) {
      newObj[key] = typeof obj[key] === 'object' ? deepClone(obj[key]) : obj[key];
    }
  }
  return newObj;
}
```


`升级版：`
``` js
const isObject = (obj) => (typeof obj === 'object' || typeof obj === 'function') && obj !== null;

const deepClone = (obj, map = new WeakMap()) => {
  // 基本类型直接返回
  if (!isObject(obj)) return obj;

  // 检查循环引用：从 map 中获取已克隆的副本
  if (map.has(obj)) {
    // 在深拷贝的逻辑里，有一个必须死守的铁律：
    // 除了基本类型（数字、字符串等），函数返回的必须是一个全新的引用地址。
    // 如果你返回 obj，你返回的是旧地址。
    // 如果你返回 map.get(obj)，你返回的是新地址。
    return map.get(obj); // 返回 map 里存的对象
  }

  let constructor = obj.constructor;

  // 普通的深拷贝逻辑是：创建一个空对象 {} -> 遍历原对象的属性 -> 赋值。
  // 如果是 Date 对象：
  // Date 对象的值并不存在于它的属性里，而是存在于它内部的一个 [[DateValue]] 槽位中。
  // 如果你用普通逻辑： for...in 循环拿不到日期的具体时间。
  // 结果： 你会得到一个空对象 {}，丢掉了所有的时间信息。
  // 如果是 RegExp 对象：
  // 正则对象包含了源码（source）和旗标（flags，如 g, i）。
  // 如果你用普通逻辑： 同样无法通过遍历属性来重新构建一个功能完整的正则。
  // 结果： 也会得到一个没有灵魂的空对象。

  // 处理正则和日期（修正正则表达式写法）
  if (/^(RegExp|Date)$/i.test(constructor.name)) {
    return new constructor(obj);
  }

  // 初始化克隆对象
  let cloneTarget = Array.isArray(obj) ? [] : {};

  // 在进入递归前，先把当前对象存入 map
  map.set(obj, cloneTarget);

  for (let prop in obj) {
    if (obj.hasOwnProperty(prop)) {
      // 递归处理子属性
      cloneTarget[prop] = deepClone(obj[prop], map);
    }
  }

  return cloneTarget;
}
```
